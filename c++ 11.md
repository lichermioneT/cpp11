**1.3初始化**

```c++
#include <iostream>
#include <string>
using namespace std;

//C++11 支持在类内部直接给非静态数据成员设置默认值
class base
{
public:
	int a{ 10 };
	double b{ 33.4 };
	char ch{ 'c'};
	string str{ "lichermionex" };
	string str2{ 10, 'a'};

// 静态的正向可以初始化在内部的,必须是正向
	static const int e1{ 55 };
	const static int e2{ 55 };

// 其它的
	static double e3;
	static char  e4;
	static const char* e5;
};

double base::e3 = 4.3;
char base::e4 = 'a';
const char* base::e5 = "fffffffff";

int main()
{
	return 0;
}
```

**C++11 允许在成员声明时直接指定默认初始值，构造函数只需要初始化与默认值不同的成员。**



**1.4 final**

**final:**

**1.修饰派生类的虚函数，表示该虚函数不能够被重写了**

**2.修饰派生类，表示该派生类不能够被重写了的、**

```c++
#include <iostream>
#include <string>
using namespace std;

class base
{
public:
	virtual void test()
	{
		cout << "base ..." << endl;
	}
};

class child final : public base
{
public:
	virtual void test() final
	{
		cout << "child ... " << endl;
	}
};

class granchind final : public base
{
public:
	virtual void test()  final
	{
		cout << "granchind ... " << endl;
	}
};

int main()
{
	child lic;
	lic.test();

	granchind gran;
	gran.test();

	return 0;
}
```

**1.9 noexcept**

```c++
#include <iostream>
#include <cassert>
using namespace std;

class myexception
{
public:
	myexception(const string& str)
		:_msg(str)
	{ }

	string _msg;
};

//void fun() throw(int, double, myexception) 指定类型
//void fun() throw() 没有类型，不允许抛异常
//void fun() noexcept(true) 控制是否抛异常 true抛异常，false不抛异常。常量表达式
void fun() noexcept   // 不允许抛异常
{
	try
	{
		//throw 1;
		throw myexception("string exception");
	}
	catch (int e)
	{
		cout << "throw value:" << e << endl;
	}
	catch (myexception e)
	{
		cout << "myexception:" << e._msg << endl;
	}

	cout << "正常执行的位置" << endl;
}

int main()
{
	fun();
	return 0;
}
```
