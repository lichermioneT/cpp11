```c++
#include <iostream>
#include <mutex>
using namespace std;

class TaskQueue
{
public:
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;
		
	static TaskQueue* getInstance()
	{	
		TaskQueue* task = _taskQueue.load();
		if (task == nullptr) 
		{
			_mutex.lock();         
			task = _taskQueue.load();
			if (task == nullptr)
			{
				task = new TaskQueue; 
				_taskQueue.store(task);
			}

			_mutex.unlock();
		}

		return task;
	}

	void print()
	{
		cout << "我是一个单例对象" << endl;
	}

private:
	TaskQueue() = default;
	static mutex _mutex;
	static atomic<TaskQueue*> _taskQueue;
};

atomic<TaskQueue*> TaskQueue::_taskQueue; // 声明一下
mutex TaskQueue::_mutex;                  // 声明一下
```

```c++
int main()
{
	TaskQueue* t1 = TaskQueue::getInstance();
	TaskQueue* t2 = TaskQueue::getInstance();
	t1->print();

	cout << t1 << endl;
	cout << t2 << endl;
	
	return 0;
}
```

```c++
#include <iostream>
#include <mutex>
#include <queue>
#include <thread>
using namespace std;

class TaskQueue
{
public:
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;
		
	static TaskQueue* getInstance()
	{	
		return _taskQueue;
	}

	// 1.判断是否为空的
	bool empty()
	{
		lock_guard<mutex> locker(_mutex);
		return _data.empty();
	}

	// 2.添加任务
	void addtask(int x)
	{
		lock_guard<mutex> locker(_mutex);
		_data.push(x);
	}

	// 3.删除任务
	bool poptask()
	{
		lock_guard<mutex> locker(_mutex);
		if (_data.empty())
		{
			return false;
		}

		_data.pop();

		return true;
	}

	// 4.取任务
	int  taketask()
	{
		lock_guard<mutex> locker(_mutex);
		if (_data.empty())
		{
			return -1;
		}

		return _data.front();
	}

private:
	TaskQueue() = default;
	static TaskQueue* _taskQueue;

	// 任务队列
	queue<int> _data;
	mutex _mutex;
};

TaskQueue* TaskQueue::_taskQueue = new TaskQueue;

int main()
{
	TaskQueue* task = TaskQueue::getInstance();

	// 生产者
	thread t1([=]()
		{
			for (int i = 0; i < 10; ++i)
			{
				task->addtask(i + 10);
				cout << "push data:" << i + 10 << "thread id :" << this_thread::get_id() << endl;
				this_thread::sleep_for(chrono::microseconds(500));
			}
		});

	// 消费者
	thread t2([=]()
		{
			this_thread::sleep_for(chrono::microseconds(100));
			while (!task->empty())
			{
				int num = task->taketask();
				cout << "take data:" << num << "thread id :" << this_thread::get_id() << endl;
				task->poptask();
				this_thread::sleep_for(chrono::microseconds(1000));
			}
		});


	t1.join();
	t2.join();

	return 0;
}
```

```c++
#include <iostream>

public:
	virtual void transform() override
	{
		cout << "变成狮子" << endl;
	}

	virtual void ability() override
	{
		cout << "ability变成狮子" << endl;
	}

	virtual ~LionSmile() override
	{

	}; // 虚析构

public:
	virtual void transform() override
	{
		cout << "变成蝙蝠" << endl;
	}

	virtual void ability() override
	{
		cout << "ability变成蝙蝠" << endl;
	}

	virtual ~BatSmile() override
	{

	}; // 虚析构

{

	}

	SmileFactory* factory = new SmileFactory;
	AbstractSmile* obj = factory->create(Type::Lion);

	return 0;
```
