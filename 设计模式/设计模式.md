**双重检查锁**

```c++
#include <iostream>
#include <mutex>
using namespace std;

class TaskQueue
{
public:
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;
		
	static TaskQueue* getInstance()
	{	
		if (_taskQueue == nullptr) // 第二次abc线程
		{
			_mutex.lock();         // 第一次 xyz线程
			if (_taskQueue == nullptr)
			{
				// 1.申请内存， 2.创建一个对象然后存储， 3，存储指针。
				// 
				// 1.申请内存， 2.存储指针， 3，创建一个对象然后存储。
				// 问题：第二步就返回了的。
				_taskQueue = new TaskQueue(); 
			}

			_mutex.unlock();
		}

		return _taskQueue;
	}

	void print()
	{
		cout << "我是一个单例对象" << endl;
	}

private:
	TaskQueue() = default;
	static TaskQueue* _taskQueue;
	static mutex _mutex;

	//TaskQueue(const TaskQueue& t) = default
	//TaskQueue& operator=(const TaskQueue& t) = default;
};

TaskQueue* TaskQueue::_taskQueue = nullptr;
mutex TaskQueue::_mutex; // 声明一下

int main()
{
	TaskQueue* t1 = TaskQueue::getInstance();

	cout << t1 << endl;
```

```c++
#include <iostream>
#include <mutex>
using namespace std;

class TaskQueue
{
public:
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;
		
	static TaskQueue* getInstance()
	{	
		TaskQueue* task = _taskQueue.load();
		if (task == nullptr) 
		{
			_mutex.lock();         
			task = _taskQueue.load();
			if (task == nullptr)
			{
				task = new TaskQueue; 
				_taskQueue.store(task);
			}

			_mutex.unlock();
		}

		return task;
	}

	void print()
	{
		cout << "我是一个单例对象" << endl;
	}

private:
	TaskQueue() = default;
	static mutex _mutex;
	static atomic<TaskQueue*> _taskQueue;
};

atomic<TaskQueue*> TaskQueue::_taskQueue; // 声明一下
mutex TaskQueue::_mutex;                  // 声明一下

int main()
{
	TaskQueue* t1 = TaskQueue::getInstance();
	TaskQueue* t2 = TaskQueue::getInstance();
	t1->print();

	cout << t1 << endl;
	cout << t2 << endl;
	
	return 0;
}
```

**案例**

```c++
#include <iostream>
#include <mutex>
#include <queue>
#include <thread>
using namespace std;

class TaskQueue
{
public:
	TaskQueue(const TaskQueue& t) = delete;
	TaskQueue& operator=(const TaskQueue& t) = delete;
		
	static TaskQueue* getInstance()
	{	
		return _taskQueue;
	}

	// 1.判断是否为空的
	bool empty()
	{
		lock_guard<mutex> locker(_mutex);
		return _data.empty();
	}

	// 2.添加任务
	void addtask(int x)
	{
		lock_guard<mutex> locker(_mutex);
		_data.push(x);
	}

	// 3.删除任务
	bool poptask()
	{
		lock_guard<mutex> locker(_mutex);
		if (_data.empty())
		{
			return false;
		}

		_data.pop();

		return true;
	}

	// 4.取任务
	int  taketask()
	{
		lock_guard<mutex> locker(_mutex);
		if (_data.empty())
		{
			return -1;
		}

		return _data.front();
	}

private:
	TaskQueue() = default;
	static TaskQueue* _taskQueue;

	// 任务队列
	queue<int> _data;
	mutex _mutex;
};

TaskQueue* TaskQueue::_taskQueue = new TaskQueue;

int main()
{
	TaskQueue* task = TaskQueue::getInstance();

	// 生产者
	thread t1([=]()
		{
			for (int i = 0; i < 10; ++i)
			{
				task->addtask(i + 10);
				cout << "push data:" << i + 10 << "thread id :" << this_thread::get_id() << endl;
				this_thread::sleep_for(chrono::microseconds(500));
			}
		});

	// 消费者
	thread t2([=]()
		{
			this_thread::sleep_for(chrono::microseconds(100));
			while (!task->empty())
			{
				int num = task->taketask();
				cout << "take data:" << num << "thread id :" << this_thread::get_id() << endl;
				task->poptask();
				this_thread::sleep_for(chrono::microseconds(1000));
			}
		});


	t1.join();
	t2.join();

	return 0;
}
```
