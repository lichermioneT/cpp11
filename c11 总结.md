**修饰函数**

```c++
#include <iostream>
#include <vector>
#include <string>
#include <map>
using namespace std;

class base
{
public:
	base(int a = 0):_a(a){ }
	
	virtual void print(char ch)
	{
		cout << "ch:" << ch << endl;
	}

	virtual void print(int x)
	{
		cout << "x:" << x << endl;
	}

private:
	int _a;
};

class D1  : public base
{
public:
	virtual void print(char ch) final  // 继承下去的函数不能够进行重写了的。
	{
		ch += 1;
		cout << "ch:" << ch << endl;
	}

	virtual void print(int x)
	{   
		x += 1;
		cout << "x:" << x << endl;
	}
};

int main()
{
	
	return 0;
}
```

**final:修饰虚函数，让继承下去的虚函数，不能够再次被复写的。**



## 5.override

```c++
#include <iostream>
#include <vector>
#include <string>
#include <map>
using namespace std;

class base
{
public:
	base(int a = 0):_a(a){ }
	
	virtual void print(char ch)
	{
		cout << "ch:" << ch << endl;
	}

	virtual void print(int x)
	{
		cout << "x:" << x << endl;
	}

private:
	int _a;
};

class D1  : public base
{
public:
	virtual void print(char ch) override
	{
		ch += 1;
		cout << "ch:" << ch << endl;
	}

	virtual void print(int x) override
	{   
		x += 1;
		cout << "x:" << x << endl;
	}
};

int main()
{
	D1 d;
	d.print(22);
	d.print('a');
	
	return 0;
}
```

**检查虚函数是否重写成功了的。**

## 1.8静态断言

```c++
#include <iostream>
using namespace std;

int main()
{
	for (int i = 0; i < 10; ++i)
	{
		cout << i << endl;
	}

	static_assert(sizeof(long long) == 8, "不是64位机器");
	
	return 0;
}
```

**静态断言：用于编译阶段，不是程序运行期间的。**
